TASK 9 - Implement and Test the Billing Program
===============================================

#include <stdio.h>
#include <stdlib.h>
#define MODULUS 65536U
#define RATE2 225LL
#define RATE3 450LL
#define FIXED 5000LL

unsigned int used(unsigned int old, unsigned int now) {
    return (now - old) % MODULUS;
}
long long bill(unsigned int u) {
    long long p = FIXED;
    if (u > 100) {
        unsigned int n = u - 100;
        if (n > 100) { p += 100*RATE2 + (long long)(n-100)*RATE3; }
        else p += (long long)n*RATE2;
    }
    return p;
}
int main(void) {
    unsigned int old, now;
    printf("Enter previous and current readings: ");
    if (scanf("%u %u",&old,&now)!=2 || old>65535 || now>65535) {
        puts("Invalid reading"); return 1;
    }
    unsigned int u=used(old,now);
    long long p=bill(u);
    printf("Units consumed: %u\n",u);
    printf("Energy + fixed total: Rs %lld.%02lld\n",p/100,p%100);
    return 0;
}
